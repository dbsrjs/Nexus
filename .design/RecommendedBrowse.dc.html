export class PreflightAbort extends Error {
constructor(reason, hint) {
super(reason);
this.name = 'PreflightAbort';
this.reason = reason;
this.hint = hint ?? null;
}
}
/** 준비 단계 호출이 실패하면 중단한다. */
export function abortUnless(ok, reason, json) {
if (ok) return;
throw new PreflightAbort(reason, hintFor(json));
}
